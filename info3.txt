import { Volume2, Volume1, VolumeX } from "lucide-react";

export default function BackgroundMusic() {
  const audioRef = useRef<HTMLAudioElement>(null);
  const mutedRef = useRef(false);
  const [playing, setPlaying] = useState(false);
  const [muted, setMuted] = useState(false);
  const [volume, setVolume] = useState(0.05);

  useEffect(() => {
    try {
      if (localStorage.getItem("myroster.music.muted") === "1") {
        mutedRef.current = true;
        setMuted(true);
      }
      const v = parseFloat(localStorage.getItem("myroster.music.volume") || "");
      if (!isNaN(v)) setVolume(v);
    } catch {}
  }, []);

  useEffect(() => {
    if (audioRef.current) audioRef.current.volume = volume;
    try { if (canStorePreferences()) localStorage.setItem("myroster.music.volume", String(volume)); } catch {}
  }, [volume]);

  const start = useCallback(async () => {
    const a = audioRef.current;
    if (!a || mutedRef.current) return false;
    try {
      await a.play();
      return true;
    } catch {
      return false;
    }
  }, []);

  // Try now; if the browser blocks it, start on the first click/touch/key
  useEffect(() => {
    start();
    const events = ["pointerdown", "keydown", "touchstart"];
    const cleanup = () => events.forEach((e) => window.removeEventListener(e, onGesture));
    const onGesture = async () => {
      if (await start()) cleanup();
    };
    events.forEach((e) => window.addEventListener(e, onGesture, { passive: true }));
    return cleanup;
  }, [start]);

  const toggleMute = async () => {
    const next = !muted;
    mutedRef.current = next;
    setMuted(next);
    try { if (canStorePreferences()) localStorage.setItem("myroster.music.muted", next ? "1" : "0"); } catch {}
    const a = audioRef.current;
    if (!a) return;
    if (next) a.pause();
    else await start();
  };

  const onButton = async () => {
    if (!playing && !muted) {
      await start();
      return;
    }
    toggleMute();
  };

  const label = muted ? "🔇 Muted" : playing ? "🔊 Music On" : "▶ Play music";

  return (
    <>
      <audio
        ref={audioRef}
        src="/aviao_es-e-mu_sicas.mp3"
        loop
        preload="auto"
        onPlaying={() => setPlaying(true)}
        onPause={() => setPlaying(false)}
      />
      <div className="fixed bottom-20 right-4 z-40 md:bottom-24 md:right-8">
        <div className="rounded-lg border border-[#333] bg-[#1a1a1a] p-3 shadow-lg backdrop-blur">
          <div className="mb-2 flex items-center gap-3">
            {muted ? <VolumeX className="h-4 w-4 text-gray-400" /> : volume < 0.3 ? <Volume1 className="h-4 w-4 text-[#4caf50]" /> : <Volume2 className="h-4 w-4 text-[#1f88ff]" />}

          </div>

          </button>

export function cruiseKnots(aircraft: string): number {
  const q = (aircraft || '').trim()
  const m = q ? matchAircraft(q).match : undefined
  const icao = (m?.icao || q).toUpperCase()
  if (KT[icao]) return KT[icao]
  const n = (m?.name || q).toLowerCase()
  if (/cessna 1|cessna 2|piper|diamond|cirrus|bonanza|mooney|baron|seneca/.test(n)) return 140
  if (/caravan|pc-12|tbm|kodiak|king air|dash 8|atr|saab|dornier/.test(n)) return 280
  return 450
}

// "0.82", ".82", "M.82" -> Mach; "450" -> knots
export function parseSpeed(s: string): number | null {
  const t = (s || '').trim().toLowerCase().replace(',', '.')
  if (!t) return null
  const m = t.match(/^m?\s*(0?\.\d+|[01]\.\d+)$/)
  if (m) return Math.round(parseFloat(m[1]) * 573)
  const n = parseFloat(t.replace(/[^0-9.]/g, ''))
  return n >= 80 && n <= 700 ? Math.round(n) : null
}

export function taxiMinutes(aircraft: string) {
  const kt = cruiseKnots(aircraft)
  return kt >= 400 ? 20 : kt >= 220 ? 14 : 8
}

// gate-to-gate minutes: taxi + climb/descent + cruise over the (routed) distance
export function estimateBlockMinutes(
  distanceNm: number,
  aircraft = '',
  opt: { speedKt?: number; routed?: boolean } = {}
): number {
  if (!distanceNm || distanceNm <= 0) return 0
  const kt = opt.speedKt || cruiseKnots(aircraft)
  const routeNm = opt.routed ? distanceNm : distanceNm * 1.04 + 20
  const avg = routeNm < 300 ? kt * 0.8 : kt
  const climbDescent = kt >= 400 ? 10 : kt >= 220 ? 6 : 3
  return Math.max(10, Math.round((routeNm / avg) * 60 + climbDescent + taxiMinutes(aircraft)))

type WP = ParsedFlightPlan['waypoints'][number]

// MSFS/SimBrief positions look like: N38° 46' 52.00",W9° 8' 3.00",+000374.00
function dms(str: string) {
  const m = Array.from(str.matchAll(/([NSEW])\s*(\d+)°\s*(\d+)'\s*([\d.]+)"/g))
  if (m.length < 2) return null
  const conv = (x: RegExpMatchArray) => {
    const v = Number(x[2]) + Number(x[3]) / 60 + Number(x[4]) / 3600
    return x[1] === 'S' || x[1] === 'W' ? -v : v
  }
  return { lat: conv(m[0]), lon: conv(m[1]) }
}

function nm(a: WP, b: WP) {
  const r = Math.PI / 180
  const dLat = (b.latitude - a.latitude) * r
  const dLon = (b.longitude - a.longitude) * r
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(a.latitude * r) * Math.cos(b.latitude * r) * Math.sin(dLon / 2) ** 2
  return 2 * 3440.07 * Math.asin(Math.sqrt(h))
}

export function parsePLNFile(xmlContent: string): ParsedFlightPlan {
  const doc = new DOMParser().parseFromString(xmlContent, 'text/xml')
  if (doc.getElementsByTagName('parsererror').length) throw new Error('Invalid .pln file')
  const first = (tag: string, root: Document | Element = doc) =>
    root.getElementsByTagName(tag)[0]?.textContent?.trim() || ''

  const waypoints: WP[] = []
  Array.from(doc.getElementsByTagName('ATCWaypoint')).forEach((el, i) => {
    const pos = dms(first('WorldPosition', el))
    if (!pos) return
    waypoints.push({ id: el.getAttribute('id') || `WP${i}`, latitude: pos.lat, longitude: pos.lon })
  })

  const departure = first('DepartureID') || waypoints[0]?.id || ''
  const arrival = first('DestinationID') || waypoints[waypoints.length - 1]?.id || ''
  if (!departure || !arrival) throw new Error('Invalid .pln file')

  let distance = 0
  for (let i = 0; i < waypoints.length - 1; i++) distance += nm(waypoints[i], waypoints[i + 1])
  distance = Math.round(distance)

  return {
    departure,
    arrival,
    aircraft: '',
    cruiseAltitude: Math.round(parseFloat(first('CruisingAlt'))) || 35000,
    distance,
    estimatedTime: distance ? Math.round((distance / 450) * 60) + 15 : 0,
    waypoints,
  }
}
