type Geo = { fromIata: string; toIata: string; fromCity: string; toCity: string; distanceNm: number }

const hhmm = (m: number) => {
  const x = ((Math.round(m) % 1440) + 1440) % 1440
  return `${String(Math.floor(x / 60)).padStart(2, '0')}:${String(x % 60).padStart(2, '0')}`
}
const inputCls = 'w-full rounded-xl border border-white/10 bg-white/5 px-3 py-2 text-sm text-white outline-none focus:border-[#1f88ff]'

function Field({ label, children, className = '' }: { label: string; children: React.ReactNode; className?: string }) {
  return (
    <label className={`block text-xs text-white/50 ${className}`}>
      {label}
      <div className="mt-1">{children}</div>
    </label>
  )
}

export default function FlightPlanPage() {
  const { id } = useParams<{ id: string }>()
  const router = useRouter()
  const pathname = usePathname()
  const [authed, setAuthed] = useState(false)
  const [seed, setSeed] = useState<any>(null)
  const [geo, setGeo] = useState<Geo | null>(null)
  const [ready, setReady] = useState(false)
  const [error, setError] = useState('')
  const [form, setForm] = useState({ flight: '', dep: '', arr: '', aircraft: 'A320', time: '12:00', fl: 350, speed: '', duration: 60 })
  const set = (k: string, v: any) => { setReady(false); setForm((f) => ({ ...f, [k]: v })) }

  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => {
      if (data.session) setAuthed(true)
      else router.replace(`/login?next=${encodeURIComponent(pathname)}`)
    })
  }, [pathname, router])

  // pre-fill when coming from a route page
  useEffect(() => {
    try {
      const raw = localStorage.getItem('myroster.routeSeed')
      if (!raw) return
      const s = JSON.parse(raw)
      if (String(s.id).toUpperCase() !== String(id).toUpperCase()) return
      setSeed(s)
      setForm((f) => ({
        ...f,
        flight: s.airlineIata || '',
        dep: s.orig || '',
        arr: s.dest || '',
        aircraft: s.aircraft || f.aircraft,
        duration: s.durationMin || f.duration,
      }))
    } catch {}
  }, [id])

  // distance and cities for whatever airports are in the form (IATA codes)
  useEffect(() => {
    const a = form.dep.trim().toUpperCase()
    const b = form.arr.trim().toUpperCase()
    if (seed && a === String(seed.orig).toUpperCase() && b === String(seed.dest).toUpperCase()) {
      setGeo({ fromIata: seed.orig, toIata: seed.dest, fromCity: seed.fromCity, toCity: seed.toCity, distanceNm: seed.distanceNm })
      return
    }
    if (!/^[A-Z]{3}$/.test(a) || !/^[A-Z]{3}$/.test(b) || a === b) { setGeo(null); return }
    const air = seed?.airlineIata || String(id).split('-')[0]
    const t = setTimeout(() => {
      fetch(`/api/route/${air}/${a}-${b}`)
        .then((r) => r.json())
        .then((d) => {
          if (d && !d.error && d.distanceNm) setGeo({ fromIata: d.from.iata, toIata: d.to.iata, fromCity: d.from.city, toCity: d.to.city, distanceNm: d.distanceNm })
          else setGeo(null)

    return () => clearTimeout(t)

const safeNext = () => {
  try {
    const n = new URLSearchParams(window.location.search).get('next') || '/'
    return n.startsWith('/') && !n.startsWith('//') ? n : '/'
  } catch {
    return '/'
  }
}

  const router = useRouter()
  const [mode, setMode] = useState<'in' | 'up'>('in')
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [msg, setMsg] = useState('')
  const [busy, setBusy] = useState(false)
  const [user, setUser] = useState<string | null>(null)
  const [needsLogin, setNeedsLogin] = useState(false)

  useEffect(() => {
    setNeedsLogin(new URLSearchParams(window.location.search).has('next'))
    supabase.auth.getUser().then(({ data }) => setUser(data.user?.email ?? null))
  }, [])

  const submit = async () => {
    setBusy(true)
    setMsg('')
    const { data, error } =
      mode === 'in'
        ? await supabase.auth.signInWithPassword({ email, password })

    if (error) return setMsg(error.message)
    if (mode === 'up' && !data.session) return setMsg('Check your email to confirm your account, then sign in.')
    router.push(safeNext())
  }

  const signOut = async () => {
    await supabase.auth.signOut()
    setUser(null)
  }

  const input = 'w-full rounded-xl border border-white/10 bg-white/5 px-3 py-3 text-sm text-white outline-none focus:border-[#1f88ff]'

    <section className="min-h-screen p-5 sm:p-8">

        {user ? (

            <h1 className="text-2xl font-semibold">Signed in</h1>

interface BoardingPassProps {

export function BoardingPass({
  airline,
  airlineIata,
  flightNumber,
  from,
  to,
  fromCity,
  toCity,
  date,
  depTime,
  boardingTime,
  arrivalTime,
  passenger,
  aircraft,
  registration,
  gate,
  seat,
  distance,
  duration,
}: BoardingPassProps) {
  const ref = useRef<HTMLDivElement>(null)

  const handleDownload = async () => {
    if (!ref.current) return
    try {
      const canvas = await html2canvas(ref.current, {
        backgroundColor: '#0a0a0a',
        scale: 2,
      })
      const link = document.createElement('a')
      link.href = canvas.toDataURL('image/png')
      link.download = `${from}-${to}-boarding-pass.png`
      link.click()
      toast.success('Boarding pass saved as image')
    } catch {
      toast.error('Failed to save image')
    }

// IATA -> ICAO lookup, built once from /api/airlines
let builtFrom: unknown = null
const iataToIcao = new Map<string, string>()

function lookupIcao(data: unknown, iata?: string) {
  if (!iata || !Array.isArray(data)) return ''
  if (builtFrom !== data) {
    const score = (x: A) => (x.active ? 1_000_000 : 0) + (x.numRoutes || 0)
    const best = new Map<string, A>()
    for (const a of data as A[]) {
      if (!a.iata || !a.icao) continue
      const k = a.iata.toUpperCase()
      const cur = best.get(k)
      if (!cur || score(a) > score(cur)) best.set(k, a)
    }
    iataToIcao.clear()
    best.forEach((a, k) => iataToIcao.set(k, String(a.icao).toUpperCase()))
    builtFrom = data
  }
  return iataToIcao.get(iata.toUpperCase()) || ''
}
