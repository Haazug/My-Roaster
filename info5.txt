type RouteHit = { airline: Airline; from: Airport; to: Airport; distanceKm: number; route: { codeshare: boolean } }
type LiveHit = { hex: string; callsign: string; reg?: string; type?: string; altFt: number | null; onGround: boolean }
type Result = {
  airlines: Airline[]
  airports: Airport[]
  routes: RouteHit[]
  flights: LiveHit[]
  aircraftTypes: { name: string; icao: string }[]
  sources: Record<string, string>
  degraded?: boolean
}

const fetcher = (url: string) => fetch(url).then((r) => r.json())
const tabs = ['All', 'Routes', 'Airlines', 'Flights', 'Airports'] as const
type Tab = (typeof tabs)[number]
const sourceLabels: Record<string, string> = { flightConnections: 'FlightConnections', flightradar24: 'Flightradar24', openSky: 'OpenSky' }

export default function SearchClient() {
  const params = useSearchParams()
  const router = useRouter()
  const pathname = usePathname()
  const [input, setInput] = useState(params.get('q') ?? '')
  const [query, setQuery] = useState(params.get('q') ?? '')
  const [tab, setTab] = useState<Tab>('All')

    const t = setTimeout(() => {
      const q = input.trim()
      setQuery(q)
      router.replace(q ? `${pathname}?q=${encodeURIComponent(q)}` : pathname, { scroll: false })
    }, 300)
    return () => clearTimeout(t)
  }, [input, pathname, router])

  const { data, isLoading, error } = useSWR<Result>(query ? `/api/search?q=${encodeURIComponent(query)}` : null, fetcher, { keepPreviousData: true })

  const counts = { Routes: data?.routes.length ?? 0, Airlines: data?.airlines.length ?? 0, Flights: data?.flights.length ?? 0, Airports: data?.airports.length ?? 0 }
  const show = (t: Exclude<Tab, 'All'>) => tab === 'All' || tab === t

  return (

        <label htmlFor="q" className="sr-only">Search</label>

        {isLoading && <Loader2 className="size-5 shrink-0 animate-spin text-white/50" aria-label="Searching" />}
      </form>

      <div className="mt-4 flex gap-2 overflow-x-auto pb-1" role="tablist">
        {tabs.map((t) => (
          <button key={t} role="tab" aria-selected={tab === t} onClick={() => setTab(t)} className={`min-h-11 shrink-0 rounded-full px-4 text-sm ${tab === t ? 'bg-[#1f88ff] text-white' : 'bg-white/10 text-white/70'}`}>

      {query && data && (

          {show('Routes') && (
            <Group title="Routes" count={data.routes.length}>
              {data.routes.map((r) => (
                <Row key={`${r.airline.id}-${r.from.iata}-${r.to.iata}`} href={`/route/${r.airline.iata || r.airline.id}/${r.from.iata}-${r.to.iata}`}>

  const [simBriefData, setSimBriefData] = useState<SimBriefData | null>(null)
  const [isLoading, setIsLoading] = useState(false)

  const parsePlnContent = (content: string): SimBriefData | null => {
    try {
      const lines = content.split('\n')
      const data: any = {}

      for (const line of lines) {
        const [key, ...valueParts] = line.split('=')
        if (key && valueParts.length > 0) {
          data[key.trim()] = valueParts.join('=').trim()
        }
      }

      return {
        origin: data['DEP'] || data['ORIGIN'] || 'UNKNOWN',
        destination: data['ARR'] || data['DESTINATION'] || 'UNKNOWN',
        airline: data['AIRLINE'] || 'UNKNOWN',
        flightNumber: data['FLIGHT'] || 'UNKNOWN',
        aircraft: data['AIRCRAFT'] || 'UNKNOWN',
        flightTime: parseInt(data['FLIGHTTIME'] || '0') || 0,
      }
    } catch (e) {
      return null
    }
  }

  const handlePaste = async (e: React.ClipboardEvent<HTMLTextAreaElement>) => {
    e.preventDefault()
    try {
      const text = await navigator.clipboard.readText()
      setPlnContent(text)
    } catch (err) {
      toast.error('Failed to read clipboard')
    }
  }

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    const reader = new FileReader()
    reader.onload = (event) => {
      const content = event.target?.result as string
      setPlnContent(content)
    }
    reader.readAsText(file)
  }

  const handleImport = () => {
    if (!plnContent.trim()) {
      toast.error('Please paste or upload a flight plan')
      return
    }

    setIsLoading(true)
    const parsed = parsePlnContent(plnContent)

    if (parsed) {
      setSimBriefData(parsed)
      toast.success('Flight plan imported successfully!')
      setTimeout(() => {
        router.push(`/flights/${id}/fly?simbrief=true`)
      }, 1500)
    } else {
      toast.error('Invalid flight plan format')
    }
    setIsLoading(false)
  }

  return (
    <section className="min-h-screen p-5 sm:p-8">
      <Link href={`/flights/${id}`} className="inline-flex min-h-11 items-center gap-2 text-sm text-white/55 hover:text-white">

type A = { id: string; name: string; iata: string; icao: string; active: boolean; numRoutes: number }
type Ap = { iata?: string; city?: string; name?: string; lat?: number; lon?: number }
type R = { from: string; to: string; fromAirport?: Ap; toAirport?: Ap; aircraft?: string[] }
type Result = { airline: A; route: R; nm: number; min: number }

const fetcher = (u: string) => fetch(u).then((r) => r.json())
const norm = (s = '') => s.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase()
const pick = <T,>(a: T[]) => a[Math.floor(Math.random() * a.length)]
const BUCKETS = [['any', 'Surprise me'], ['short', 'Under 2h'], ['medium', '2–5h'], ['long', '5h+']] as const

function nmBetween(a?: Ap, b?: Ap) {
  if (a?.lat == null || a?.lon == null || b?.lat == null || b?.lon == null) return 0
  const r = Math.PI / 180
  const h = Math.sin(((b.lat - a.lat) * r) / 2) ** 2 + Math.cos(a.lat * r) * Math.cos(b.lat * r) * Math.sin(((b.lon - a.lon) * r) / 2) ** 2
  return Math.round(2 * 3440.07 * Math.asin(Math.sqrt(h)))
}

export default function RandomPage() {

  const generate = async () => {
    setBusy(true); setMsg(''); setRes(null)
    const airline = chosen || suggestions[0] || null
    const pool = airline ? [airline] : list.filter((a) => a.active && a.numRoutes > 0).sort(() => Math.random() - 0.5).slice(0, 15)
    const na = norm(airport.trim())
    for (const a of pool) {
      const d = await fetch(`/api/airlines/${a.id}`).then((r) => r.json()).catch(() => null)
      const routes: R[] = d?.routes || []
      const ok: Result[] = []
      for (const r of routes) {

  } catch {
    return '/'

}
