const hash = (s: string) => {
  let h = 2166136261
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i)
    h = Math.imul(h, 16777619)
  }
  return h >>> 0
}
const rng = (seed: number) => () => {
  seed = (Math.imul(seed, 1664525) + 1013904223) >>> 0
  return seed / 4294967296
}

function Barcode({ seed }: { seed: string }) {
  const r = rng(hash(seed))
  const W = 140, H = 36, rows = 6, rh = H / rows
  const out: React.ReactElement[] = []
  const bars = (x0: number, x1: number, k: string) => {
    let x = x0, on = true, i = 0
    while (x < x1) {
      const w = Math.min(1 + Math.floor(r() * 3), x1 - x)
      if (on) out.push(<rect key={`${k}${i++}`} x={x} y={0} width={w} height={H} />)
      x += w
      on = !on
    }
  }
  bars(0, 16, 'l')
  bars(W - 16, W, 'r')
  for (let row = 0; row < rows; row++) {
    let x = 20, on = r() > 0.5, i = 0
    while (x < W - 20) {
      const w = Math.min(1 + Math.floor(r() * 4), W - 20 - x)
      if (on) out.push(<rect key={`m${row}-${i++}`} x={x} y={row * rh} width={w} height={rh} />)
      x += w
      on = !on
    }
  }
  return (
    <svg viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" className="h-20 w-full text-white" fill="currentColor" aria-hidden="true">
      {out}
    </svg>
  )
}

export const CONSENT_KEY = 'myroster.consent'
export type Consent = { v: 1; preferences: boolean; ts: number }

export function getConsent(): Consent | null {
  try {
    const raw = window.localStorage.getItem(CONSENT_KEY)
    return raw ? (JSON.parse(raw) as Consent) : null
  } catch {
    return null
  }
}

export function setConsent(preferences: boolean) {
  try {
    window.localStorage.setItem(CONSENT_KEY, JSON.stringify({ v: 1, preferences, ts: Date.now() }))
    if (!preferences) {
      window.localStorage.removeItem('myroster.music.volume')
      window.localStorage.removeItem('myroster.music.muted')
    }
  } catch {}
}

import { useCallback, useEffect, useState } from "react"
import { supabase } from "@/lib/supabase"

const SYNCED = ["myroster.logs", "myroster.favourites", "myroster.bookings", "myroster.customRoutes"]

const merge = (a: any[], b: any[]) => {
  const m = new Map<string, any>()
  ;[...b, ...a].forEach((x, i) => m.set(String(x?.id ?? i), x))
  return Array.from(m.values())
}

async function push(key: string, value: unknown) {
  if (!SYNCED.includes(key)) return
  try {
    const { data } = await supabase.auth.getSession()
    const uid = data.session?.user.id
    if (!uid) return
    await supabase.from("user_data").upsert({ user_id: uid, key, value, updated_at: new Date().toISOString() })
  } catch {}
}

export function useLocalStorage<T>(key: string, initial: T) {
  const [value, setValue] = useState<T>(initial)
  const [hydrated, setHydrated] = useState(false)

  useEffect(() => {
    let local: any = null
    try { const raw = window.localStorage.getItem(key); if (raw) { local = JSON.parse(raw); setValue(local) } } catch {}
    setHydrated(true)
    if (!SYNCED.includes(key)) return
    let cancelled = false
    ;(async () => {
      try {
        const { data: s } = await supabase.auth.getSession()
        const uid = s.session?.user.id
        if (!uid) return
        const { data, error } = await supabase.from("user_data").select("value").eq("user_id", uid).eq("key", key).maybeSingle()
        if (error || cancelled) return
        const mark = `myroster.synced.${uid}.${key}`
        const first = !window.localStorage.getItem(mark)
        const cloud = Array.isArray(data?.value) ? (data!.value as any[]) : null
        const mine = Array.isArray(local) ? local : []
        // first sync on this device merges; afterwards the account is the source of truth
        const next = cloud === null ? mine : first ? merge(mine, cloud) : cloud
        window.localStorage.setItem(key, JSON.stringify(next))
        window.localStorage.setItem(mark, "1")
        if (!cancelled) setValue(next as T)
        if (cloud === null || first) push(key, next)
      } catch {}
    })()
    return () => { cancelled = true }
  }, [key])

  const update = useCallback((next: T | ((current: T) => T)) => {
    setValue(current => {
      const resolved = typeof next === "function" ? (next as (current: T) => T)(current) : next
      try { window.localStorage.setItem(key, JSON.stringify(resolved)) } catch {}
      push(key, resolved)
      return resolved
    })
  }, [key])

  return [value, update, hydrated] as const
}
